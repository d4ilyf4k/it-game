import test from 'node:test';import assert from 'node:assert/strict';import * as E from '../dist/engine.js';
function graduate(id='frontend'){const s=E.createGame(id);for(let i=0;i<4;i++){E.answerQuiz(s,E.quizQuestion(s).answer);E.nextQuiz(s);}E.startBoss(s);for(let i=0;i<3;i++){E.answerBoss(s,E.bossQuestion(s).answer);E.nextBoss(s);}return s;}
function shift(s){if(s.pending)E.chooseEvent(s,0);if(s.focus<20)E.recover(s);E.answerTask(s,E.task(s).answer);E.endDay(s);}
test('five distinct professions each unlock career only after boss',()=>{assert.equal(E.ROLES.length,5);for(const r of E.ROLES){const s=graduate(r.id);assert.equal(s.stage,'career');assert.equal(s.xp,40);assert.equal(s.money,700);assert.throws(()=>E.nextBoss(s));}});
test('quiz feedback cannot award duplicate response',()=>{const s=E.createGame();E.answerQuiz(s,0);assert.throws(()=>E.answerQuiz(s,0));assert.equal(s.quizAnswers.length,1);});
test('boss times out, loses lives and can restart',()=>{const s=E.createGame();for(let i=0;i<4;i++){E.answerQuiz(s,0);E.nextQuiz(s);}E.startBoss(s);E.tickBoss(s,121);assert.equal(s.stage,'bossFailed');E.startBoss(s);assert.equal(s.bossRemaining,120);const q=E.bossQuestion(s),wrong=q.options.map((_,i)=>i).filter(i=>i!==q.answer);E.answerBoss(s,wrong[0]);assert.equal(s.bossHp,2);assert.throws(()=>E.answerBoss(s,wrong[0]));E.answerBoss(s,wrong[1]);E.answerBoss(s,q.answer);E.nextBoss(s);E.answerBoss(s,(E.bossQuestion(s).answer+1)%3);assert.equal(s.stage,'bossFailed');});
test('task and contract reject duplicate payouts',()=>{const s=graduate();E.answerTask(s,E.task(s).answer);const money=s.money;assert.throws(()=>E.answerTask(s,0));assert.equal(s.money,money);});
test('wrong task can be retried, recovery avoids softlock',()=>{const s=graduate();const q=E.task(s);E.answerTask(s,(q.answer+1)%3);assert.equal(s.taskDone,false);s.focus=0;assert.throws(()=>E.answerTask(s,q.answer));E.recover(s);E.answerTask(s,q.answer);assert.equal(s.taskDone,true);});
test('shop gates by rank and ownership and applies bonuses',()=>{const s=graduate();assert.throws(()=>E.buy(s,'special'));E.buy(s,'chair');assert.throws(()=>E.buy(s,'chair'));s.focus=0;E.act(s,'rest');assert.equal(s.focus,40);});
test('hiring and expansion impossible without owning company',()=>{const s=graduate();assert.throws(()=>E.hire(s,'maya'));assert.throws(()=>E.expand(s));assert.throws(()=>E.acquire(s,'Test'));});
test('career progression and company complete for every profession',()=>{for(const r of E.ROLES){const s=graduate(r.id);let cap=0;while(E.rank(s)<4||s.money<3800){shift(s);assert.ok(cap++<40);}E.acquire(s,'Студия теста');E.hire(s,'liza');while(!s.finished){if(s.focus<20)E.recover(s);E.answerTask(s,E.task(s).answer);E.contract(s,E.role(s).questions[7+s.company.contracts%3].answer);if(s.company.level<3&&s.money>s.company.level*1000+700)E.expand(s);for(const c of E.CANDIDATES)if(!s.company.staff.some(e=>e.id===c.id)&&s.company.staff.length<s.company.level*2&&s.money>c.fee+700)E.hire(s,c.id);E.endDay(s);assert.ok(cap++<60);}assert.equal(s.company.staff.length,6);assert.equal(E.restore(JSON.stringify(s)).finished,true);console.log(r.id,'company completed on day',s.day);}});
test('payroll charged and unaffordable end day blocked',()=>{const s=graduate();s.xp=1050;s.money=5000;E.acquire(s,'X');E.hire(s,'maya');E.answerTask(s,E.task(s).answer);s.money=0;assert.throws(()=>E.endDay(s));E.advance(s);const before=s.money;E.endDay(s);assert.equal(s.money,before-90);});
test('company capacity and once-a-day contract',()=>{const s=graduate();s.xp=1050;s.money=10000;E.acquire(s,'X');E.hire(s,'maya');E.hire(s,'timur');assert.throws(()=>E.hire(s,'liza'));E.contract(s,E.role(s).questions[7].answer);assert.throws(()=>E.contract(s,0));});
test('director prioritizes stress and respects pending event',()=>{const s=graduate();s.stress=90;E.tickWorld(s,46);assert.equal(s.pending,'burnout');E.tickWorld(s,60);assert.equal(s.worldSeconds,46);E.chooseEvent(s,0);assert.equal(s.stress,65);});
test('save roundtrip and corrupt state validation',()=>{const s=graduate('security');assert.equal(E.restore(JSON.stringify(s)).role,'security');s.inventory=['unknown'];assert.throws(()=>E.restore(JSON.stringify(s)));assert.throws(()=>E.restore('{"version":1}'));});
test('leaderboard has own profession and employee counts',()=>{const s=graduate('sysadmin');const row=E.leaderboard(s).find(r=>r.you);assert.equal(row.profession,'Системный администратор');assert.equal(row.employees,0);});
test('practical task pays once and checks focus and career stage',()=>{const s=graduate();const before=s.money;const result=E.completePracticalTask(s);assert.equal(s.money,before+result.reward);assert.equal(s.taskDone,true);assert.throws(()=>E.completePracticalTask(s));assert.throws(()=>E.completePracticalTask(E.createGame()));const tired=graduate();tired.focus=0;assert.throws(()=>E.completePracticalTask(tired));assert.equal(tired.taskDone,false);});
test('practical company contract requires owner and staff and pays once',()=>{const s=graduate();assert.throws(()=>E.completePracticalContract(s));s.money=6000;s.xp=1050;E.acquire(s,'Практика');assert.throws(()=>E.completePracticalContract(s));E.hire(s,'liza');const before=s.money;E.completePracticalContract(s);assert.equal(s.money,before+500);assert.throws(()=>E.completePracticalContract(s));});
